# ===== 替换原来的 token 提取段（原逻辑：从 memory/2026-08-14.md 正则提取）=====
# 新逻辑：从 Windows 凭据管理器读取，失败则提示手动注入

function Get-GhToken {
    # 优先：环境变量（CI/临时注入）
    if ($env:GH_TOKEN) { return $env:GH_TOKEN }
    # 其次：凭据管理器
    try {
        $cred = Get-StoredCredential -Target "GitHubPagesToken" -ErrorAction Stop
        if ($cred) {
            # SecureString 转 plaintext（仅内存，不落盘）
            return [Runtime.InteropServices.Marshal]::PtrToStringAuto(
                [Runtime.InteropServices.Marshal]::SecureStringToBSTR($cred.Password)
            )
        }
    } catch { }
    # 兜底：报错退出，不读文件
    Write-Error "未找到 token。请先执行: cmdkey /generic:GitHubPagesToken /user:token /pass:ghp_xxx  或  `$env:GH_TOKEN='ghp_xxx'"
    exit 1
}

$Token = Get-GhToken

## 完成标准（四问阻塞门）

每个任务收尾四问，任一不满足 = 不算完成：

1. **结果验证了吗？**（测试 / 检查 / 截图 / 输出）
2. **记忆沉淀了吗？**（observations / events.log / 每日笔记 —— 见第4问）
3. **该 commit 吗？**（重要变更 git commit）
4. **本次沉淀了吗？（阻塞门 R2-强制）**
   - 必须写 ≥1 条记录，否则任务标记为「未完成」：
     - 失败类 → events.log（带反思三字段：原因 / 尝试 / 下次）
     - 成功类 → observations.md（标 positive + 可复用模式）
     - 用户偏好/纠正 → observations.md（标 source=user）
   - 跳过此问 = 违规，phoenix-triggers.md 视为单点阻塞，立即停下补写

> 第4问是自进化的原料入口。没有沉淀，梦境循环无东西可凝练，规则永远不晋升。
> 一条也算，流水账不算——必须带「下次怎么做」或「可复用模式」。

# scripts/event-log-hook.ps1 — 工具失败自动写 events.log
# 用法：在 exec 命令后接管道调用，或包成 wrapper
# 示例：Invoke-Command { 你的命令 } | Tee-Object result.txt ; if($LASTEXITCODE -ne 0){ .\scripts\event-log-hook.ps1 -Cmd "你的命令" -ExitCode $LASTEXITCODE }

param(
    [Parameter(Mandatory)][string]$Cmd,
    [Parameter(Mandatory)][int]$ExitCode,
    [string]$Note = ""
)
$ts = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$entry = "[$ts] [FAIL] exit=$ExitCode cmd=$Cmd"
if ($Note) { $entry += " note=$Note" }
$entry += "  反思: 原因= 待填 / 尝试= 待填 / 下次= 待填"

# UTF-8 无 BOM 追加（规避 PowerShell 5.1 GBK 坑）
$logPath = "memory/events.log"
[System.IO.File]::AppendAllText((Resolve-Path $logPath).Path, $entry + "`n", (New-Object System.Text.UTF8Encoding $false))
Write-Host "[hook] 已记录失败到 events.log，请补全反思三字段" -ForegroundColor Yellow

# scripts/memory-health-check.ps1 — 周度记忆体检
# 挂 cron 每周一 9:00 跑，统计上周沉淀率，低于阈值向用户告警

$week = (Get-Date).AddDays(-7)
$msg = @()

# 统计 events.log 本周失败条数
$events = Get-Content memory/events.log -ErrorAction SilentlyContinue | Where-Object { $_ -match (Get-Date -Format "yyyy-MM-dd") }
$failCount = ($events | Where-Object { $_ -match "FAIL" }).Count
$msg += "本周 events.log 失败条数: $failCount"

# 统计 observations 本周新增
$obs = Get-Content memory/observations.md -ErrorAction SilentlyContinue
$obsCount = ($obs | Select-String "^\## \[" + (Get-Date -Format "yyyy-MM-dd")).Count
$msg += "本周 observations 新增: $obsCount"

# 统计 episodes 本周新增
$epi = Get-Content memory/episodes.md -ErrorAction SilentlyContinue
$epiCount = ($epi | Select-String "^\## \[" + (Get-Date -Format "yyyy-MM-dd")).Count
$msg += "本周 episodes 新增: $epiCount"

# 候选层积压告警
$pending = ($obs | Select-String "状态.*候选").Count
if ($pending -gt 20) {
    $msg += "⚠️ 候选层积压 $pending 条（>20），请跑一轮 evolution-cycle.sh --dry 做晋升"
}

# 阈值：本周交付物数 vs 沉淀条数
$total = $failCount + $obsCount + $epiCount
if ($total -lt 3) {
    $msg += "⚠️ 本周沉淀仅 $total 条，低于阈值3。检查 path-routing.md 第4问是否在执行"
}

$output = $msg -join "`n"
[System.IO.File]::AppendAllText("memory/evolution-log.md", "`n## [$(Get-Date -Format 'yyyy-MM-dd HH:mm')] 记忆体检`n$output`n", (New-Object System.Text.UTF8Encoding $false))
if ($total -lt 3 -or $pending -gt 20) { Write-Host $output -ForegroundColor Red }
else { Write-Host $output -ForegroundColor Green }
